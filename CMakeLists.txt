cmake_minimum_required(VERSION 3.24)
project(Racer LANGUAGES CXX OBJCXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(SDL3 3.4 CONFIG REQUIRED)
set(BGFX_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(BGFX_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BGFX_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BGFX_INSTALL OFF CACHE BOOL "" FORCE)
foreach(dep bgfx bx bimg)
 string(TOUPPER ${dep} upper)
 set(${upper}_DIR "${CMAKE_SOURCE_DIR}/.deps/${dep}" CACHE PATH "" FORCE)
endforeach()
add_subdirectory(.deps/bgfx.cmake EXCLUDE_FROM_ALL)
add_library(racer_sim src/simulation.cpp src/setup.cpp src/profile.cpp src/test_session.cpp src/play_flow.cpp src/presentation.cpp src/diagnostics.cpp src/livery.cpp src/parts.cpp src/document_store.cpp src/career.cpp src/garage_features.cpp)
target_include_directories(racer_sim PUBLIC src)
target_sources(racer_sim PRIVATE src/livery_raster.cpp)
target_sources(racer_sim PRIVATE src/career_flow.cpp)
add_library(racer_assets src/mesh_asset.cpp)
add_library(racer_geometry src/vehicle_geometry.cpp)
target_include_directories(racer_geometry PUBLIC src)
target_link_libraries(racer_sim PUBLIC racer_geometry)
target_sources(racer_assets PRIVATE src/vehicle_asset.cpp)
target_include_directories(racer_assets PUBLIC src PRIVATE third_party/cgltf)
target_link_libraries(racer_assets PRIVATE bimg_decode bimg bx PUBLIC racer_geometry)
add_library(racer_input src/gamepad.cpp)
target_link_libraries(racer_input PUBLIC racer_sim SDL3::SDL3)
file(GLOB identity_sources LIST_DIRECTORIES false CONFIGURE_DEPENDS "src/*" "scripts/*" "tests/*" "third_party/cgltf/*")
list(APPEND identity_sources "${CMAKE_SOURCE_DIR}/CMakeLists.txt" "${CMAKE_SOURCE_DIR}/references/dependencies.md")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${identity_sources})
set(identity_material "${CMAKE_CXX_COMPILER_ID}-${CMAKE_CXX_COMPILER_VERSION}-${CMAKE_BUILD_TYPE}")
foreach(source IN LISTS identity_sources)
  file(SHA256 "${source}" digest)
  string(APPEND identity_material "${digest}")
endforeach()
string(SHA256 RACER_BUILD_ID "${identity_material}")
add_executable(racer src/main.cpp)
target_compile_definitions(racer PRIVATE RACER_BUILD_ID="${RACER_BUILD_ID}")
target_compile_definitions(racer PRIVATE RACER_VEHICLE_LIBRARY="${CMAKE_SOURCE_DIR}/assets/local/vehicles/catalog.racer")
target_link_libraries(racer PRIVATE racer_input racer_assets SDL3::SDL3 bgfx bx)
target_include_directories(racer PRIVATE .deps/bgfx/examples/common/debugdraw)
target_compile_definitions(racer PRIVATE RACER_SHADER_DIR="${CMAKE_SOURCE_DIR}/.deps/bgfx/examples/runtime/shaders/metal")
enable_testing()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME diagnostic_query_checks COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/diagnostic_query_checks.py)
add_test(NAME slrr_asset_checks COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/slrr_asset_checks.py)
add_test(NAME slrr_resource_checks COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/slrr_resource_checks.py)
add_test(NAME slrr_gltf_material_checks COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/slrr_gltf_material_checks.py)
add_test(NAME juiced2_data_checks COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/juiced2_data_checks.py)
add_test(NAME juiced2_wheel_contract_checks COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/juiced2_wheel_contract_checks.py)
add_test(NAME build_preservation_checks COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/build_preservation_checks.py)
foreach(check car_import car_install prepared_install car_coverage research_inventory prepared_inventory vehicle_capture vehicle_selection feedback_capture verification_lifecycle calibration gt4_preparation gt4_metadata mcla_preparation ford_vehicle ford_cohort ford_registry slrr_vehicle slrr_resource_literal car_scale juiced2_vehicle juiced2_registry juiced2_cohort slrr_preparation mc3_repair mount_fit vehicle_track slrr_clearance vehicle_clearance)
  add_test(NAME ${check}_checks COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/${check}_checks.py)
endforeach()
add_executable(mesh_asset_checks tests/mesh_asset_checks.cpp)
add_test(NAME mcla_player_checks COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/mcla_player_checks.py)
set_tests_properties(mcla_player_checks PROPERTIES SKIP_RETURN_CODE 77)
foreach(check mcla_expansion juiced2_tyre_fit juiced2_template mc3_preparation mc3_wheel_contract mc3_cohort ford_static_rest mc3_cohort_glass mc3_cohort_exhaust car_stages juiced2_stock_tyre_fit juiced2_wheel_state juiced2_native_sizing verify_selection_reload)
  add_test(NAME ${check}_checks COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/${check}_checks.py)
  set_tests_properties(${check}_checks PROPERTIES SKIP_RETURN_CODE 77)
endforeach()
target_link_libraries(mesh_asset_checks PRIVATE racer_assets)
add_test(NAME mesh_asset_checks COMMAND mesh_asset_checks)
add_executable(vehicle_geometry_checks tests/vehicle_geometry_checks.cpp)
target_link_libraries(vehicle_geometry_checks PRIVATE racer_geometry bx)
add_test(NAME vehicle_geometry_checks COMMAND vehicle_geometry_checks)
add_executable(vehicle_asset_checks tests/vehicle_asset_checks.cpp)
target_link_libraries(vehicle_asset_checks PRIVATE racer_assets racer_sim)
add_test(NAME vehicle_asset_checks COMMAND vehicle_asset_checks)
add_executable(vehicle_validate src/vehicle_validate.cpp)
target_link_libraries(vehicle_validate PRIVATE racer_assets)
add_test(NAME vehicle_validator_cli_checks COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/vehicle_validator_cli_checks.py $<TARGET_FILE:vehicle_validate>)
add_executable(material_shader_checks tests/material_shader_checks.cpp)
target_include_directories(material_shader_checks PRIVATE src .deps/bgfx/examples/common/debugdraw)
add_test(NAME material_shader_checks COMMAND material_shader_checks)
add_executable(render_checks tests/render_checks.cpp)
target_link_libraries(render_checks PRIVATE racer_sim bgfx bx)
add_test(NAME render_checks COMMAND render_checks)
add_executable(driving_checks tests/driving_checks.cpp)
target_link_libraries(driving_checks PRIVATE racer_sim)
add_test(NAME driving_checks COMMAND driving_checks)

add_executable(gamepad_checks tests/gamepad_checks.cpp)
target_link_libraries(gamepad_checks PRIVATE racer_input)
add_test(NAME gamepad_checks COMMAND gamepad_checks)
add_executable(review_script_checks tests/review_script_checks.cpp)
target_link_libraries(review_script_checks PRIVATE racer_input)
add_test(NAME review_script_checks COMMAND review_script_checks)

foreach(check setup session flow presentation diagnostic livery livery_raster parts document_store career career_flow career_integration garage_feature vehicle_config performance_class workflow_observation)
  add_executable(${check}_checks tests/${check}_checks.cpp)
  target_link_libraries(${check}_checks PRIVATE racer_sim)
  add_test(NAME ${check}_checks COMMAND ${check}_checks)
